Read any spectrometer file, convert it for your software.

nirs4all-formats is a chemometrics library that reads the spectroscopy file zoo into one clean data model and writes it back for market software. This page is the library itself, compiled to WebAssembly: drop a file to read its spectra, metadata and provenance, convert it, and export it. Nothing leaves your machine.

~58 format families · core version HDF5 in-wasm no server · no upload

Drop a spectroscopy file anywhere

OPUS · JCAMP-DX · SPC · ASD · CSV · HDF5 · PerkinElmer · …

One file for most formats. For multi-file formats (ENVI, …) drop the data file and its sidecars together.

Convert · export

From any file to your software

Read a file above, pick the software or format you need, and the page shows what the target requires, plans the conversion, computes it with nirs4all-methods, lets you compare the result with the original and writes the file, listing everything the target cannot carry.

  1. 1

    Read

    Drop a file or load a sample. Every signal keeps its own axis, units, metadata and provenance.

  2. 2

    Plan

    nirs4all-formats states what the target needs and the steps to get there, asks what the file does not say and refuses, with a fix, what cannot be converted.

  3. 3

    Convert

    nirs4all-methods computes each step in your browser: reflectance from references, absorbance, wavelength axes, known grids.

  4. 4

    Check

    The converted spectra appear next to the originals, before and after, with clamped points, missing values and grids reported.

  5. 5

    Export

    The writer lists its losses and records the conversions in the file when the format holds text.

    crates.io
    // cargo add nirs4all-formats
use nirs4all_formats::{open_path, plan_conversion, write_records, PlanOptions, WriteOptions};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    // Read: the format is recognised from the file content
    let records = open_path("barley.csv")?;

    // Access the data: one record per spectrum, each signal owns its axis
    let signal = &records[0].signals["signal"];
    println!("{:?} in {}", signal.signal_type, signal.axis.unit);
    let matrix: Vec<&[f64]> = records
        .iter()
        .map(|record| record.signals["signal"].values.as_slice())
        .collect();
    println!("{:?} {:?}", records[0].targets, records[0].metadata.get("sample_id"));

    // Export for a software or a format; what it cannot carry is listed
    let output = write_records(&records, "winisi", &WriteOptions::default())?;
    output.save("exported", false)?;
    for loss in &output.losses {
        println!("{}: {}", loss.code, loss.message);
    }

    // Plan a conversion: nirs4all-methods computes each step
    let raw = open_path("sample.ProcSpec")?;
    let plan = plan_conversion(&raw, Some("winisi"), &PlanOptions::default())?;
    println!("{:?}: {} steps, {} spectra", plan.status, plan.steps.len(), matrix.len());
    Ok(())
}

    // C ABI archive (include/, lib/) from GitHub Releases, or cargo build -p nirs4all-formats-capi
// cc example.c -Iinclude -Llib -Wl,-rpath,"$PWD/lib" -lnirs4all_formats_capi && ./a.out
#include <stdio.h>
#include "nirs4all_formats.h"

// Print a result (or the error) and free it
static void show(n4fmt_status_t status, char *json, char *error) {
    printf("%.160s\n", status == N4FMT_STATUS_OK ? json : error);
    n4fmt_string_free(json);
    n4fmt_string_free(error);
}

int main(void) {
    char *records = NULL, *json = NULL, *error = NULL;

    // Read: the format is recognised from the file content
    n4fmt_status_t status = n4fmt_open_path("barley.csv", NULL, &records, &error);
    if (status != N4FMT_STATUS_OK) {
        show(status, NULL, error);
        return 1;
    }

    // Access the data: a JSON array of records, one per spectrum, for any JSON parser:
    // [{"signals": {"signal": {"axis": {"values": [...], "unit": "nm"}, "values": [...]}},
    //   "targets": {"Protein": 9.383, ...}, "metadata": {"sample_id": "1", ...}, "provenance": ...}]
    printf("%.160s\n", records);

    // Export for a software or a format: base64 files, and what the target cannot carry
    status = n4fmt_write_records(records, "winisi", "{\"base_name\": \"barley\"}", &json, &error);
    show(status, json, error);
    status = n4fmt_convert_path("barley.csv", "jcamp-dx", "exported", NULL, &json, &error);
    show(status, json, error);
    status = n4fmt_export_targets(&json, &error);   // formats, software presets, requirements
    show(status, json, error);
    n4fmt_string_free(records);

    // Plan a conversion: steps, questions and refusals;
    // nirs4all-methods computes the values
    n4fmt_open_path("sample.ProcSpec", NULL, &records, &error);
    status = n4fmt_plan_conversion(records, "winisi", NULL, &json, &error);
    show(status, json, error);
    n4fmt_string_free(records);
    return 0;
}

import java.nio.file.Path;
import java.util.List;
import org.nirs4all.formats.*;

public class Example {
    public static void main(String[] args) {
        // Read: the format is recognised from the file content
        List<SpectralRecord> records = Nirs4allFormats.open(Path.of("barley.csv"));

        // Access the data
        SignalMatrix data = SignalMatrix.stack(records, "signal");
        double[][] x = data.values();                    // spectra matrix (n x p)
        double[] axis = data.axis().values();            // data.axis().unit(): "nm"
        SpectralRecord record = records.get(0);
        Signal signal = record.signal("signal");         // signal.signalType(): "absorbance"
        double protein = record.numericTarget("Protein");
        String sample = record.metadata().get("sample_id").asText();
        String format = record.provenance().format();    // "delimited-text"

        // Export for a software or a format; what it cannot carry is listed
        WriteResult out = Nirs4allFormats.write(records, "winisi");   // FOSS WinISI .nir
        System.out.println(out.files().get(0).name() + " " + out.losses());
        Nirs4allFormats.convert(Path.of("barley.csv"), "jcamp-dx", Path.of("exported"));
        Nirs4allFormats.exportTargets().software();      // OPUS, OMNIC, WinISI, Unscrambler...

        // Plan a conversion: steps, questions and refusals;
        // nirs4all-methods computes the values
        ConversionPlan plan = Nirs4allFormats.planConversion(Nirs4allFormats.open(Path.of("sample.ProcSpec")), "winisi");
        System.out.println(plan.status() + " " + plan.steps().stream().map(ConversionPlan.Step::methodId).toList());
    }
}
